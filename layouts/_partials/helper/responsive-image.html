{{- /* Params: Resource: Hugo image resource object (Optional, for processing) Widths: Slice of widths to generate for srcset Attributes: Map of HTML attributes for the tag Emits a element with AVIF and WebP sources plus the original format as fallback when the resource can be processed; otherwise emits a plain . */ -}} {{- $resource := .Resource -}} {{- $widths := .Widths -}} {{- $attributes := .Attributes | default dict -}} {{- $avifSrcset := slice -}} {{- $webpSrcset := slice -}} {{- $originalSrcset := slice -}} {{- if and $widths $resource (reflect.IsImageResourceProcessable $resource) -}} {{- range $widths -}} {{- if lt . $resource.Width -}} {{- $resized := $resource.Resize (printf "%dx" .) -}} {{- $originalSrcset = $originalSrcset | append (printf "%s %dw" $resized.RelPermalink .) -}} {{- $avifSrcset = $avifSrcset | append (printf "%s %dw" ($resized.Process "avif q50").RelPermalink .) -}} {{- $webpSrcset = $webpSrcset | append (printf "%s %dw" ($resized.Process "webp q82").RelPermalink .) -}} {{- end -}} {{- end -}} {{- if gt (len $originalSrcset) 0 -}} {{- $permalink := default $resource.RelPermalink (index $attributes "src") -}} {{- $originalSrcset = $originalSrcset | append (printf "%s %dw" $permalink $resource.Width) -}} {{- $avifSrcset = $avifSrcset | append (printf "%s %dw" ($resource.Process "avif q50").RelPermalink $resource.Width) -}} {{- $webpSrcset = $webpSrcset | append (printf "%s %dw" ($resource.Process "webp q82").RelPermalink $resource.Width) -}} {{- $attributes = merge $attributes (dict "srcset" (delimit $originalSrcset ", ")) -}} {{- end -}} {{- end -}} {{- $img := printf "" (partial "helper/attributes" $attributes) -}} {{- if gt (len $webpSrcset) 0 -}} {{ $img | safeHTML }} {{- else -}} {{ $img | safeHTML }} {{- end -}}
